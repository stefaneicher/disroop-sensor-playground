4static int16_t get_i16(const uint8_t *p)
5{
6	return (int16_t)(uint16_t)(p[0] | (p[1] << 8));
7}

9icm_frame_status_t icm45686_parse_frame(const uint8_t *in, size_t len, icm_sample_t *out)
10{
11	if (in == NULL || out == NULL || len < ICM_FIFO_FRAME_LEN) {
12		return ICM_FRAME_SHORT;
13	}
14	const uint8_t hdr = in[0];
15	if ((hdr & (ICM_FIFO_HDR_EXT | ICM_FIFO_HDR_20BIT)) ||
16	    (hdr & (ICM_FIFO_HDR_ACCEL | ICM_FIFO_HDR_GYRO)) != (ICM_FIFO_HDR_ACCEL | ICM_FIFO_HDR_GYRO)) {
17		return ICM_FRAME_INVALID;
18	}
19	for (int i = 0; i < 3; i++) {
20		const int16_t a = get_i16(&in[1 + 2 * i]);
21		const int16_t g = get_i16(&in[7 + 2 * i]);
22		if (a == ICM_INVALID_SAMPLE || g == ICM_INVALID_SAMPLE) {
23			return ICM_FRAME_INVALID;
24		}
25		out->accel_g[i] = (float)a / ICM_ACCEL_LSB_PER_G;
26		out->gyro_dps[i] = (float)g / ICM_GYRO_LSB_PER_DPS;
27		/* 0x8000 ist "ungueltig", daher Saettigung bei +-32767 */
28		out->accel_sat[i] = (a == INT16_MAX || a == -INT16_MAX);
29		out->gyro_sat[i] = (g == INT16_MAX || g == -INT16_MAX);
30	}
31	out->timestamp = (uint16_t)(in[14] | (in[15] << 8));
32	return ICM_FRAME_OK;
33}

35static int wr(const icm_bus_t *bus, uint8_t reg, uint8_t v)
36{
37	return bus->write(bus->ctx, reg, &v, 1);
38}

40icm_status_t icm45686_ireg_write(const icm_bus_t *bus, uint16_t addr, uint8_t value)
41{
42	/* hal_tdk write_sreg: Adresse High/Low + Daten in einem Burst, danach >= 4 us warten */
43	const uint8_t buf[3] = {(uint8_t)(addr >> 8), (uint8_t)addr, value};
44	if (bus->write(bus->ctx, ICM_REG_IREG_ADDR_15_8, buf, sizeof buf) != 0) {
45		return ICM_ERR_BUS;
46	}
47	bus->delay_us(bus->ctx, 4);
48	return ICM_OK;
49}

51static icm_status_t configure_measurement(const icm_bus_t *bus, uint8_t odr)
52{
53	int e = 0;
54	e |= wr(bus, ICM_REG_TMST_WOM_CONFIG, 0);
55	e |= wr(bus, ICM_REG_INT1_CONFIG1, 0);
56	e |= wr(bus, ICM_REG_ACCEL_CONFIG0, ICM_ACCEL_FS_32G | odr);
57	e |= wr(bus, ICM_REG_GYRO_CONFIG0, ICM_GYRO_FS_4000DPS | odr);
58	e |= wr(bus, ICM_REG_INT1_CONFIG0, ICM_INT1_FIFO_THS);
59	e |= wr(bus, ICM_REG_PWR_MGMT0, ICM_PWR_ACCEL_LN | ICM_PWR_GYRO_LN);
60	if (e) {
61		return ICM_ERR_BUS;
62	}
63	bus->delay_us(bus->ctx, 70000); /* Gyro-Startzeit 70 ms laut hal_tdk */
64	return ICM_OK;
65}

67icm_status_t icm45686_init(const icm_bus_t *bus, uint8_t odr, uint16_t watermark_frames)
68{
69	if (wr(bus, ICM_REG_MISC2, ICM_MISC2_SOFT_RST)) {
70		return ICM_ERR_BUS;
71	}
72	bus->delay_us(bus->ctx, 1000);
73	uint8_t who = 0;
74	if (bus->read(bus->ctx, ICM_REG_WHO_AM_I, &who, 1)) {
75		return ICM_ERR_BUS;
76	}
77	if (who != ICM_WHO_AM_I_VALUE) {
78		return ICM_ERR_WHOAMI;
79	}
80	int e = 0;
81	e |= wr(bus, ICM_REG_INT1_CONFIG2, ICM_INT1_PUSH_PULL_PULSED_HIGH);
82	e |= wr(bus, ICM_REG_INT2_CONFIG2, ICM_INT1_PUSH_PULL_PULSED_HIGH); /* INT2 push-pull (PINOUT), ungenutzt */
83	e |= wr(bus, ICM_REG_FIFO_CONFIG3, 0);
84	e |= wr(bus, ICM_REG_FIFO_CONFIG0, ICM_FIFO_MODE_STREAM | ICM_FIFO_DEPTH_MAX);
85	e |= wr(bus, ICM_REG_FIFO_CONFIG1_0, (uint8_t)watermark_frames);
86	e |= wr(bus, ICM_REG_FIFO_CONFIG1_1, (uint8_t)(watermark_frames >> 8));
87	e |= wr(bus, ICM_REG_FIFO_CONFIG2, ICM_FIFO_FLUSH | ICM_FIFO_WM_GT_TH);
88	e |= wr(bus, ICM_REG_FIFO_CONFIG3, ICM_FIFO_IF_EN | ICM_FIFO_ACCEL_EN | ICM_FIFO_GYRO_EN);
89	if (e) {
90		return ICM_ERR_BUS;
91	}
92	return configure_measurement(bus, odr);
93}

95icm_status_t icm45686_fifo_count(const icm_bus_t *bus, uint16_t *frames)
96{
97	uint8_t b[2];
98	if (bus->read(bus->ctx, ICM_REG_FIFO_COUNT_0, b, 2)) {
99		return ICM_ERR_BUS;
100	}
101	*frames = (uint16_t)(b[0] | (b[1] << 8));
102	return ICM_OK;
103}

105icm_status_t icm45686_enter_wom(const icm_bus_t *bus, uint16_t threshold_mg)
106{
107	uint32_t lsb = ((uint32_t)threshold_mg * 256u + 500u) / 1000u;
108	if (lsb > 255u) {
109		lsb = 255u;
110	}
111	int e = 0;
112	e |= wr(bus, ICM_REG_INT1_CONFIG0, 0);
113	e |= wr(bus, ICM_REG_PWR_MGMT0, 0);
114	e |= wr(bus, ICM_REG_ACCEL_CONFIG0, ICM_ACCEL_FS_32G | ICM_ODR_25HZ);
115	if (e) {
116		return ICM_ERR_BUS;
117	}
118	for (uint16_t i = 0; i < 3; i++) {
119		if (icm45686_ireg_write(bus, (uint16_t)(ICM_IREG_ACCEL_WOM_X_THR + i), (uint8_t)lsb) != ICM_OK) {
120			return ICM_ERR_BUS;
121		}
122	}
123	e |= wr(bus, ICM_REG_TMST_WOM_CONFIG, ICM_WOM_MODE_PREV);
124	e |= wr(bus, ICM_REG_PWR_MGMT0, ICM_PWR_ACCEL_LP);
125	if (e) {
126		return ICM_ERR_BUS;
127	}
128	bus->delay_us(bus->ctx, 10000); /* Accel-Startzeit 10 ms */
129	e |= wr(bus, ICM_REG_TMST_WOM_CONFIG, ICM_WOM_EN | ICM_WOM_MODE_PREV);
130	e |= wr(bus, ICM_REG_INT1_CONFIG1, ICM_INT1_WOM_XYZ);
131	return e ? ICM_ERR_BUS : ICM_OK;
132}

134icm_status_t icm45686_exit_wom(const icm_bus_t *bus, uint8_t odr)
135{
136	if (wr(bus, ICM_REG_FIFO_CONFIG2, ICM_FIFO_FLUSH | ICM_FIFO_WM_GT_TH)) {
137		return ICM_ERR_BUS;
138	}
139	return configure_measurement(bus, odr);
