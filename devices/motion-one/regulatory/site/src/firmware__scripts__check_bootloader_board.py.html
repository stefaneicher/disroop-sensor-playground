1#!/usr/bin/env python3
2"""Gleicht die Adafruit-Bootloader-Board-Definition (bootloader/motion_one/board.h) gegen
3electronics/PINOUT_V1.json ab. Exit 0 nur wenn: RGB-Status-LED = LED_R/G/B, LED aktiv low,
4DFU-Taster = BTN mit internem Pull-up, kein zweiter (OTA-)Taster auf fremdem Pin, keine UART-Pins (Q-02)."""
5import json, re, sys, pathlib
6
7FW = pathlib.Path(__file__).resolve().parents[1]
8pinout = pathlib.Path(sys.argv[1]) if len(sys.argv) > 1 else FW.parent / "electronics/PINOUT_V1.json"
9board = pathlib.Path(sys.argv[2]) if len(sys.argv) > 2 else FW / "bootloader/motion_one/board.h"
10errs = []
11if not board.exists():
12    print(f"FAIL {board} fehlt"); print("bootloader-pincheck FAIL"); sys.exit(1)
13data = json.loads(pinout.read_text())
14pins = {}
15for p in data["pins"]:
16    m = re.match(r"P([01])\.(\d+)", p["gpio"])
17    if m:
18        pins[p["signal"]] = (int(m[1]), int(m[2]))
19func = {p["signal"]: p["function"].lower() for p in data["pins"]}
20text = re.sub(r"//.*|/\*.*?\*/", "", board.read_text(), flags=re.S)
21defs = {m[1]: m[2].strip() for m in re.finditer(r"^[ \t]*#define[ \t]+(\w+)[ \t]+(.+?)[ \t]*$", text, re.M)}
22
23def pin(name):
24    v = defs.get(name)
25    m = v and re.fullmatch(r"PINNUM\(\s*(\d)\s*,\s*(\d+)\s*\)", v)
26    return (int(m[1]), int(m[2])) if m else None
27
28for macro, sig in (("LED_RGB_RED_PIN", "LED_R"), ("LED_RGB_GREEN_PIN", "LED_G"),
29                   ("LED_RGB_BLUE_PIN", "LED_B"), ("LED_PRIMARY_PIN", "LED_R"), ("BUTTON_DFU", "BTN")):
30    got = pin(macro)
31    if got != pins.get(sig):
32        errs.append(f"{macro}: board.h {got} != PINOUT {sig} {pins.get(sig)}")
33if "aktiv low" in func.get("LED_R", "") and defs.get("LED_STATE_ON") != "0":
34    errs.append("LED_STATE_ON muss 0 sein (aktiv low)")
35if defs.get("LEDS_NUMBER") != "1" or "LED_SECONDARY_PIN" in defs:
36    errs.append("genau eine Primaer-LED (RGB), keine Sekundaer-LED")
37if "pull-up" in func.get("BTN", "") and defs.get("BUTTON_PULL") != "NRF_GPIO_PIN_PULLUP":
38    errs.append("BUTTON_PULL muss NRF_GPIO_PIN_PULLUP sein")
39if "BUTTON_DFU_OTA" in defs:
40    errs.append("BUTTON_DFU_OTA definiert, V1 hat nur einen Taster")
41if any(re.search(r"UART|_TX_PIN|_RX_PIN", k) for k in defs):
42    errs.append("UART-Definition vorhanden (Q-02: kein UART)")
43used = {v for k in defs if (v := pin(k))}
44own = set(pins.values())
45for k in defs:
46    if (v := pin(k)) and v not in own:
47        errs.append(f"{k}: Pin P{v[0]}.{v[1]} nicht im PINOUT")
48print(f"PINOUT {pinout.name} revision {data.get('revision')}: board.h {len(used)} Pins geprueft")
49for e in errs:
50    print("FAIL", e)
51print("bootloader-pincheck", "FAIL" if errs else "PASS")
52sys.exit(1 if errs else 0)
