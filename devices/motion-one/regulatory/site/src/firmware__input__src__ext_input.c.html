4bool ext_board_present(uint16_t mv_pullup, uint16_t mv_pulldown)
5{
6	/* offener Eingang folgt dem Pull (~3.3 V / ~0 V); Stick-Wiper (Quelle <= 2.5 kOhm) bleibt nahe beisammen */
7	const uint16_t d = mv_pullup > mv_pulldown ? mv_pullup - mv_pulldown : mv_pulldown - mv_pullup;
8	return d < EXT_PROBE_FLOAT_MV;
9}

18bool stick_axis_cal_center(stick_axis_cal_t *c, const uint16_t *s, size_t n)
19{
20	if (n == 0) {
21		return false;
22	}
23	uint32_t sum = 0;
24	for (size_t i = 0; i < n; i++) {
25		sum += s[i];
26	}
27	const uint32_t m = (sum + n / 2) / n;
28	if (m < EXT_FULL_MV * 30u / 100u || m > EXT_FULL_MV * 70u / 100u) {
29		return false;
30	}
31	c->center_mv = (uint16_t)m;
32	return true;
33}

35int8_t stick_axis_scale(const stick_axis_cal_t *c, uint16_t mv, uint8_t deadzone_pct)
36{
37	if (mv > c->max_mv) {
38		mv = c->max_mv;
39	}
40	if (mv < c->min_mv) {
41		mv = c->min_mv;
42	}
43	int32_t pct;
44	if (mv >= c->center_mv) {
45		const int32_t span = (int32_t)c->max_mv - c->center_mv;
46		pct = span > 0 ? ((int32_t)mv - c->center_mv) * 100 / span : 0;
47	} else {
48		const int32_t span = (int32_t)c->center_mv - c->min_mv;
49		pct = span > 0 ? ((int32_t)mv - c->center_mv) * 100 / span : 0;
50	}
51	const int32_t a = pct < 0 ? -pct : pct;
52	if (deadzone_pct >= 100u || a < deadzone_pct) {
53		return 0;
54	}
55	const int32_t r = (a - deadzone_pct) * 100 / (100 - deadzone_pct);
56	return (int8_t)(pct < 0 ? -r : r);
57}

61uint32_t stick_next_poll_ms(stick_poll_t *p, int8_t x, int8_t y, bool active_input, uint32_t now_ms)
62{
63	if (x != 0 || y != 0 || active_input) {
64		p->last_active_ms = now_ms;
65		p->ever_active = true;
66		return EXT_POLL_ACTIVE_MS;
67	}
68	if (p->ever_active && now_ms - p->last_active_ms <= EXT_ACTIVE_HOLD_MS) {
69		return EXT_POLL_ACTIVE_MS;
70	}
71	return EXT_POLL_IDLE_MS;
72}

76touch_evt_t touch_update(touch_t *t, bool level, uint32_t now_ms)
77{
78	if (level != t->raw) {
79		t->raw = level;
80		t->t_change = now_ms;
81		if (t->locked) {
82			t->lock_until = now_ms + TOUCH_LOCKOUT_MS; /* Sperre laeuft erst nach 10 s Ruhe ab */
83			return TOUCH_EVT_NONE;
84		}
85		if (t->edges == 0 || now_ms - t->win_start >= 1000u) {
86			t->win_start = now_ms;
87			t->edges = 0;
88		}
89		if (++t->edges > TOUCH_SPAM_EDGES) {
90			t->locked = true;
91			t->lock_until = now_ms + TOUCH_LOCKOUT_MS;
92			if (t->stable) {
93				t->stable = false;
94				return TOUCH_EVT_UP; /* kein haengendes "gedrueckt" */
95			}
96		}
97		return TOUCH_EVT_NONE;
98	}
99	if (t->locked) {
100		if ((int32_t)(now_ms - t->lock_until) < 0) {
101			return TOUCH_EVT_NONE;
102		}
103		t->locked = false; /* still entsperren: aktueller Pegel gilt als Ruhezustand */
104		t->stable = level;
105		t->edges = 0;
106		return TOUCH_EVT_NONE;
107	}
108	if (t->raw != t->stable && now_ms - t->t_change >= TOUCH_DEBOUNCE_MS) {
109		t->stable = t->raw;
110		return t->stable ? TOUCH_EVT_DOWN : TOUCH_EVT_UP;
111	}
112	return TOUCH_EVT_NONE;
113}
