30static int read_mv(const struct adc_dt_spec *ch, uint16_t *mv)
31{
32	int16_t raw;
33	struct adc_sequence seq = {.buffer = &raw, .buffer_size = sizeof raw};
34	int rc = adc_sequence_init_dt(ch, &seq);
35	rc = rc ? rc : adc_read_dt(ch, &seq);
36	int32_t v = raw;
37	rc = rc ? rc : adc_raw_to_millivolts_dt(ch, &v);
38	*mv = rc || v < 0 ? 0 : (uint16_t)MIN(v, UINT16_MAX);
39	return rc;
40}

42/* EXT1 mit internem Pull-up bzw. Pull-down (13 kOhm) messen: offen folgt dem Pull, Stick nicht. */
43static bool probe_board(void)
44{
45	uint16_t up = 0, dn = 0;
46	(void)gpio_pin_configure(gpio0, PIN_EXT1, GPIO_INPUT | GPIO_PULL_UP);
47	k_msleep(1);
48	(void)read_mv(&ax, &up);
49	(void)gpio_pin_configure(gpio0, PIN_EXT1, GPIO_INPUT | GPIO_PULL_DOWN);
50	k_msleep(1);
51	(void)read_mv(&ax, &dn);
52	(void)gpio_pin_configure(gpio0, PIN_EXT1, GPIO_DISCONNECTED);
53	LOG_INF("J5-Probe EXT1 up %u mV, down %u mV", up, dn);
54	return ext_board_present(up, dn);
55}

58static K_WORK_DELAYABLE_DEFINE(stick_work, stick_poll);

60static void stick_poll(struct k_work *w)
61{
62	ARG_UNUSED(w);
63	if (!ble_is_connected()) {
64		k_work_schedule(&stick_work, K_SECONDS(1)); /* keine ADC-Abtastung ohne Abnehmer */
65		return;
66	}
67	uint16_t mx = 0, my = 0;
68	(void)read_mv(&ax, &mx);
69	(void)read_mv(&ay, &my);
70	const int8_t x = stick_axis_scale(&cx, mx, EXT_DEADZONE_PCT);
71	const int8_t y = stick_axis_scale(&cy, my, EXT_DEADZONE_PCT);
72	const bool btn = gpio_pin_get_dt(&ext4) > 0;
73	static int8_t lx, ly;
74	static bool lb;
75	if (x != lx || y != ly || btn != lb) {
76		lx = x; ly = y; lb = btn;
77		ext_publish_stick(x, y, btn);
78	}
79	k_work_schedule(&stick_work, K_MSEC(stick_next_poll_ms(&poll, x, y, btn || touch.stable, k_uptime_get_32())));
80}

82static void touch_eval(struct k_work *w);
83static K_WORK_DELAYABLE_DEFINE(touch_work, touch_eval);
84
85static void touch_eval(struct k_work *w)
86{
87	ARG_UNUSED(w);
88	const touch_evt_t e = touch_update(&touch, gpio_pin_get_dt(&ext5) > 0, k_uptime_get_32());
89	if (e != TOUCH_EVT_NONE) {
90		ext_publish_touch(e == TOUCH_EVT_DOWN);
91	} else if (!!(ui.flags & UI_EXT_TOUCH_LOCKED) != touch.locked) {
92		ext_publish_touch(touch.stable); /* Sperrstatus melden */
93	}
94	if (touch.locked || touch.raw != touch.stable) {
95		k_work_reschedule(&touch_work, K_MSEC(touch.locked ? 1000 : TOUCH_DEBOUNCE_MS));
96	}
97}

99static void touch_isr(const struct device *d, struct gpio_callback *cb, uint32_t pins)
100{
101	ARG_UNUSED(d); ARG_UNUSED(cb); ARG_UNUSED(pins);
102	(void)touch_update(&touch, gpio_pin_get_dt(&ext5) > 0, k_uptime_get_32()); /* Flanke zaehlen */
103	k_work_reschedule(&touch_work, K_MSEC(TOUCH_DEBOUNCE_MS));
104}

106int ext_input_start(void)
107{
108	stick_axis_cal_init(&cx);
109	stick_axis_cal_init(&cy);
110	touch_init(&touch);
111	if (!adc_is_ready_dt(&ax) || !gpio_is_ready_dt(&ext4) || !gpio_is_ready_dt(&ext5)) {
112		return -ENODEV;
113	}
114	int rc = adc_channel_setup_dt(&ax);
115	rc = rc ? rc : adc_channel_setup_dt(&ay);
116	rc = rc ? rc : gpio_pin_configure_dt(&ext4, GPIO_INPUT);
117	rc = rc ? rc : gpio_pin_configure_dt(&ext5, GPIO_INPUT);
118	if (rc) {
119		return rc;
120	}
121	present = probe_board();
122	if (present) {
123		uint16_t sx[8], sy[8]; /* Mitte kalibrieren: Stick beim Start in Ruhe (ANNAHME) */
124		for (int i = 0; i < 8; i++) {
125			(void)read_mv(&ax, &sx[i]);
126			(void)read_mv(&ay, &sy[i]);
127		}
128		center_ok = stick_axis_cal_center(&cx, sx, 8) & stick_axis_cal_center(&cy, sy, 8);
129		if (!center_ok) {
130			LOG_WRN("Stick-Mitte unplausibel, Default 1.65 V");
131		}
132		k_work_schedule(&stick_work, K_MSEC(EXT_POLL_IDLE_MS));
133	} else {
134		(void)gpio_pin_configure_dt(&ext4, GPIO_DISCONNECTED); /* kein Pull-up-Strom ohne Platine */
135	}
136	/* Touch auch ohne Stick: EXT5 hat Pull-down, offen -> low, keine Flanken */
137	gpio_init_callback(&touch_cb, touch_isr, BIT(ext5.pin));
138	(void)gpio_add_callback(ext5.port, &touch_cb);
139	(void)gpio_pin_interrupt_configure_dt(&ext5, GPIO_INT_EDGE_BOTH);
140	LOG_INF("J5 Bedienplatine %s", present ? "erkannt" : "nicht erkannt");
141	return 0;
142}

153static void ui_tx(struct k_work *w);
154static K_WORK_DELAYABLE_DEFINE(ui_tx_work, ui_tx);
155
156static void ui_tx(struct k_work *w)
157{
158	ARG_UNUSED(w);
159	thingy_motion_config_t cfg;
160	ble_get_motion_config(&cfg);
161	uint32_t retry = 0;
162	if (ui_ext_rate_poll(&rate, cfg.motion_freq_hz, k_uptime_get_32(), &retry)) {
163		ui_ext_next_seq(&ui);
164		ble_publish_ui_ext(&ui);
165	} else if (retry) {
166		k_work_reschedule(&ui_tx_work, K_MSEC(retry)); /* Flanke spaetestens nach <= 100 ms */
167	}
168}

170static void ui_set(uint8_t flags, int8_t x, int8_t y)
171{
172	flags |= present ? UI_EXT_PRESENT : 0u;
173	flags |= touch.locked ? UI_EXT_TOUCH_LOCKED : 0u;
174	flags |= center_ok ? UI_EXT_CENTER_OK : 0u;
175	const bool changed = flags != ui.flags || x != ui.x || y != ui.y;
176	ui.flags = flags;
177	ui.x = x;
178	ui.y = y;
179	ui_ext_rate_mark(&rate, changed);
180	if (changed) {
181		k_work_schedule(&ui_tx_work, K_NO_WAIT);
182	}
183}

185static void ui_sub(struct k_work *w)
186{
187	ARG_UNUSED(w);
188	ui_set(ui.flags & (UI_EXT_BTN | UI_EXT_TOUCH), ui.x, ui.y);
189	ui_ext_rate_mark(&rate, true);
190	k_work_reschedule(&ui_tx_work, K_NO_WAIT);
191}
192static K_WORK_DEFINE(ui_sub_work, ui_sub);

196void ext_publish_stick(int8_t x, int8_t y, bool btn)
197{
198	ui_set((uint8_t)((ui.flags & UI_EXT_TOUCH) | (btn ? UI_EXT_BTN : 0u)), x, y);
199}
