1# Host-Unit-Tests (gcc + Unity) im Image disroop-motion-one-fw: make test
2UNITY_DIR ?= /opt/unity
3BUILD ?= build/host
4CFLAGS = -std=c11 -Wall -Wextra -Werror -O2 -Icodec/include -Ifusion/include -Iicm/include -Ipower/include -Iinput/include -I$(UNITY_DIR)/src
5UNITY = $(UNITY_DIR)/src/unity.c
6HDRS = $(wildcard codec/include/*.h fusion/include/*.h icm/include/*.h power/include/*.h input/include/*.h)
7
8TESTS = test_thingy_codec test_thingy_profile test_motion_fusion test_icm45686 test_vbat test_motor_drive test_ext_input test_ui_ext
9SRC_test_thingy_codec = codec/src/thingy_codec.c
10SRC_test_thingy_profile = codec/src/thingy_codec.c
11SRC_test_motion_fusion = fusion/src/motion_fusion.c
12SRC_test_icm45686 = icm/src/icm45686.c
13SRC_test_vbat = power/src/vbat.c
14SRC_test_motor_drive = power/src/motor_drive.c
15SRC_test_ext_input = input/src/ext_input.c
16SRC_test_ui_ext = codec/src/thingy_codec.c
17
18.SECONDEXPANSION:
19$(BUILD)/%: tests/%.c $$(SRC_%) $(HDRS)
20	mkdir -p $(BUILD)
21	$(CC) $(CFLAGS) -o $@ $< $(SRC_$*) $(UNITY) -lm
22
23# Alle Tests laufen, auch wenn einer fehlschlaegt; Exit != 0, wenn einer fehlschlaegt.
24test: $(addprefix $(BUILD)/,$(TESTS))
25	@rc=0; echo "== pincheck"; python3 scripts/check_pinout.py || rc=1; echo "== bootloader-pincheck"; python3 scripts/check_bootloader_board.py || rc=1; for t in $(TESTS); do echo "== $$t"; ./$(BUILD)/$$t || rc=1; done; exit $$rc
26
27clean:
28	rm -rf $(BUILD)
29
30pincheck:
31	python3 scripts/check_pinout.py
32	python3 scripts/check_bootloader_board.py
33
34.PHONY: test clean pincheck
