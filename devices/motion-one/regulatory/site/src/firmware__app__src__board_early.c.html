1/* Frueheste Pin-Initialisierung (PCB-Hinweis 2026-11-07): IMU-CS sofort high (kein Pull-up auf V1),
2 * VBAT_EN low (Teiler aus). Laeuft in PRE_KERNEL_1 Prioritaet 0, vor GPIO- und SPI-Treiber. */
3#include <zephyr/init.h>
4#include <zephyr/devicetree.h>
5#include <hal/nrf_gpio.h>
6
7#define CS_SPEC DT_GPIO_PIN_BY_IDX(DT_NODELABEL(spi3), cs_gpios, 0)
8#define CS_PORT DT_PROP(DT_GPIO_CTLR_BY_IDX(DT_NODELABEL(spi3), cs_gpios, 0), port)
9#define EN_PIN DT_GPIO_PIN(DT_PATH(zephyr_user), vbat_en_gpios)
10#define EN_PORT DT_PROP(DT_GPIO_CTLR(DT_PATH(zephyr_user), vbat_en_gpios), port)
11
12static int board_early_init(void)
13{
14	const uint32_t cs = NRF_GPIO_PIN_MAP(CS_PORT, CS_SPEC);
15	const uint32_t en = NRF_GPIO_PIN_MAP(EN_PORT, EN_PIN);
16	nrf_gpio_pin_set(cs);
17	nrf_gpio_cfg_output(cs);
18	nrf_gpio_pin_clear(en);
19	nrf_gpio_cfg_output(en);
20	return 0;
21}
22
23SYS_INIT(board_early_init, PRE_KERNEL_1, 0);
