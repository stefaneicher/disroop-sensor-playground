25static void led_rgb(uint8_t r, uint8_t g, uint8_t b)
26{
27	const uint8_t v[3] = {r, g, b};
28	for (int i = 0; i < 3; i++) {
29		(void)pwm_set_pulse_dt(&led[i], (uint32_t)((uint64_t)led[i].period * v[i] / 255u));
30	}
31}

33static void led_mix(uint8_t mix, uint32_t level255)
34{
35	led_rgb((mix & 1) ? level255 : 0, (mix & 2) ? level255 : 0, (mix & 4) ? level255 : 0);
36}

38static void led_anim(struct k_work *w);
39static K_WORK_DELAYABLE_DEFINE(led_work, led_anim);

41static void led_anim(struct k_work *w)
42{
43	ARG_UNUSED(w);
44	const uint32_t max = led_cur.intensity_pct * 255u / 100u;
45	if (led_cur.mode == THINGY_LED_ONE_SHOT) {
46		led_mix(led_cur.color_mix, 0);
47		return;
48	}
49	/* Dreieck auf/ab, Periode delay_ms */
50	const uint32_t s = breathe_step++ % (2 * BREATHE_STEPS);
51	const uint32_t lvl = (s < BREATHE_STEPS ? s : 2 * BREATHE_STEPS - s) * max / BREATHE_STEPS;
52	led_mix(led_cur.color_mix, lvl);
53	k_work_schedule(&led_work, K_MSEC(MAX(led_cur.delay_ms / (2 * BREATHE_STEPS), 1u)));
54}

56void ui_led_apply(const thingy_led_cmd_t *cmd)
57{
58	(void)k_work_cancel_delayable(&led_work);
59	led_cur = *cmd;
60	switch (cmd->mode) {
61	case THINGY_LED_CONSTANT:
62		led_rgb(cmd->r, cmd->g, cmd->b);
63		break;
64	case THINGY_LED_BREATHE:
65		breathe_step = 0;
66		k_work_schedule(&led_work, K_NO_WAIT);
67		break;
68	case THINGY_LED_ONE_SHOT:
69		led_mix(cmd->color_mix, cmd->intensity_pct * 255u / 100u);
70		k_work_schedule(&led_work, K_MSEC(1000)); /* ANNAHME 1 s Blitz */
71		break;
72	default:
73		led_rgb(0, 0, 0);
74		break;
75	}
76}

78/* F-12 Arbitrierung (power/motor_drive.c): Summer hat Vorrang, Motor pausiert waehrend Ton (VBUS <= 500 mA). */
79static K_MUTEX_DEFINE(load_lock);
80static bool buz_req, mot_req;
81
82static void motor_output(bool on)
83{
84	uint32_t pulse = 0;
85	if (on) {
86		const bool usb = nrf_power_usbregstatus_vbusdet_get(NRF_POWER);
87		const uint16_t duty = motor_duty_permille(motor_supply_mv(battery_last_mv(), usb));
88		pulse = (uint32_t)((uint64_t)motor.period * duty / 1000u);
89	}
90	(void)pwm_set_pulse_dt(&motor, pulse);
91}
92
93static void loads_apply(void)
94{
95	const load_out_t o = load_arbitrate(buz_req, mot_req);
96	motor_output(o.motor_on);
97}

99static void buzzer_off(struct k_work *w)
100{
101	ARG_UNUSED(w);
102	k_mutex_lock(&load_lock, K_FOREVER);
103	(void)pwm_set_dt(&buzzer, buzzer.period, 0);
104	buz_req = false;
105	loads_apply(); /* pausierten Motor fortsetzen */
106	k_mutex_unlock(&load_lock);
107	ble_publish_speaker_done();
108}
109static K_WORK_DELAYABLE_DEFINE(buzzer_work, buzzer_off);

111void ui_speaker_tone(const thingy_speaker_tone_t *t)
112{
113	if (t->freq_hz == 0 || t->volume_pct == 0 || t->duration_ms == 0) {
114		k_work_reschedule(&buzzer_work, K_NO_WAIT);
115		return;
116	}
117	const uint32_t period = PWM_HZ(t->freq_hz);
118	/* Lautstaerke ueber Tastgrad, max. 50 % (ANNAHME, Summer-Kennlinie unbekannt) */
119	const uint32_t pulse = (uint32_t)((uint64_t)period * t->volume_pct / 200u);
120	k_mutex_lock(&load_lock, K_FOREVER);
121	buz_req = true;
122	loads_apply(); /* Motor zuerst aus, dann Summer an */
123	(void)pwm_set_dt(&buzzer, period, pulse);
124	k_mutex_unlock(&load_lock);
125	k_work_reschedule(&buzzer_work, K_MSEC(t->duration_ms));
126}

150static void btn_poll(struct k_work *w)
151{
152	ARG_UNUSED(w);
153	const bool pressed = gpio_pin_get_dt(&btn) > 0;
154	static bool last;
155	if (pressed != last) {
156		last = pressed;
157		ble_publish_button(pressed);
158	}
159	switch (thingy_button_update(&btn_sm, pressed, k_uptime_get_32())) {
160	case THINGY_BTN_EVT_LONG:
161		imu_request_calibration();
162		ui_vibrate_ms(150);
163		break;
164	default:
165		break;
166	}
167	if (pressed) {
168		k_work_schedule(&btn_work, K_MSEC(BTN_POLL_MS));
169	}
170}

182int ui_init(void)
183{
184	thingy_button_init(&btn_sm);
185	for (int i = 0; i < 3; i++) {
186		if (!pwm_is_ready_dt(&led[i])) {
187			return -ENODEV;
188		}
189	}
190	if (!pwm_is_ready_dt(&buzzer) || !gpio_is_ready_dt(&btn) || !pwm_is_ready_dt(&motor)) {
191		return -ENODEV;
192	}
193	(void)pwm_set_pulse_dt(&motor, 0);
194	(void)gpio_pin_configure_dt(&btn, GPIO_INPUT);
195	gpio_init_callback(&btn_cb, btn_isr, BIT(btn.pin));
196	(void)gpio_add_callback(btn.port, &btn_cb);
197	(void)gpio_pin_interrupt_configure_dt(&btn, GPIO_INT_EDGE_BOTH);
198	led_rgb(0, 0, 0);
199	return 0;
