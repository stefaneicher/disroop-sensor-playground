1/* VBAT-Messung ueber schaltbaren Teiler (PINOUT_V1: VBAT_EN P0.05, VBAT_ADC AIN2, 100k/100k).
2 * Reines C ohne Zephyr; Hardware ueber Port. */
3#ifndef VBAT_H
4#define VBAT_H
5
6#include <stdint.h>
7
8/* QUELLE PINOUT_V1.json/PCB-Handoff 2026-11-07: Freigabe >= 25 ms vor Messung (tau 5 ms, 5 tau -> < 1 %). */
9#define VBAT_SETTLE_MS 25u
10
11typedef struct {
12	void *ctx;
13	int (*set_enable)(void *ctx, int on);
14	void (*sleep_ms)(void *ctx, uint32_t ms);
15	int (*read_adc_mv)(void *ctx, int32_t *mv);
16} vbat_port_t;
17
18/* adc_mv * num / den, geklemmt auf 0..UINT16_MAX; den == 0 -> 0. */
19uint16_t vbat_scale_mv(int32_t adc_mv, uint32_t num, uint32_t den);
20
21/* EN an, >= VBAT_SETTLE_MS warten, ADC lesen, EN immer aus. 0 = ok; Fehler -> *mv_out = 0. */
22int vbat_measure(const vbat_port_t *p, uint32_t num, uint32_t den, uint16_t *mv_out);
23
24#endif
