8#define DEG2RAD 0.017453292519943295f
9#define RAD2DEG 57.29577951308232f

11void madgwick_init(madgwick_t *f, float beta)
12{
13	f->q[0] = 1.0f;
14	f->q[1] = f->q[2] = f->q[3] = 0.0f;
15	f->beta = beta;
16}

18void madgwick_update(madgwick_t *f, const float gyro_dps[3], const float accel_g[3], float dt_s)
19{
20	float q0 = f->q[0], q1 = f->q[1], q2 = f->q[2], q3 = f->q[3];
21	const float gx = gyro_dps[0] * DEG2RAD, gy = gyro_dps[1] * DEG2RAD, gz = gyro_dps[2] * DEG2RAD;
22	float ax = accel_g[0], ay = accel_g[1], az = accel_g[2];
23
24	float qd0 = 0.5f * (-q1 * gx - q2 * gy - q3 * gz);
25	float qd1 = 0.5f * (q0 * gx + q2 * gz - q3 * gy);
26	float qd2 = 0.5f * (q0 * gy - q1 * gz + q3 * gx);
27	float qd3 = 0.5f * (q0 * gz + q1 * gy - q2 * gx);
28
29	const float an = sqrtf(ax * ax + ay * ay + az * az);
30	if (an > 1e-6f) {
31		ax /= an;
32		ay /= an;
33		az /= an;
34		const float _2q0 = 2.0f * q0, _2q1 = 2.0f * q1, _2q2 = 2.0f * q2, _2q3 = 2.0f * q3;
35		const float _4q0 = 4.0f * q0, _4q1 = 4.0f * q1, _4q2 = 4.0f * q2;
36		const float _8q1 = 8.0f * q1, _8q2 = 8.0f * q2;
37		const float q0q0 = q0 * q0, q1q1 = q1 * q1, q2q2 = q2 * q2, q3q3 = q3 * q3;
38		float s0 = _4q0 * q2q2 + _2q2 * ax + _4q0 * q1q1 - _2q1 * ay;
39		float s1 = _4q1 * q3q3 - _2q3 * ax + 4.0f * q0q0 * q1 - _2q0 * ay - _4q1 + _8q1 * q1q1 +
40			   _8q1 * q2q2 + _4q1 * az;
41		float s2 = 4.0f * q0q0 * q2 + _2q0 * ax + _4q2 * q3q3 - _2q3 * ay - _4q2 + _8q2 * q1q1 +
42			   _8q2 * q2q2 + _4q2 * az;
43		float s3 = 4.0f * q1q1 * q3 - _2q1 * ax + 4.0f * q2q2 * q3 - _2q2 * ay;
44		const float sn = sqrtf(s0 * s0 + s1 * s1 + s2 * s2 + s3 * s3);
45		if (sn > 1e-9f) {
46			qd0 -= f->beta * s0 / sn;
47			qd1 -= f->beta * s1 / sn;
48			qd2 -= f->beta * s2 / sn;
49			qd3 -= f->beta * s3 / sn;
50		}
51	}
52	q0 += qd0 * dt_s;
53	q1 += qd1 * dt_s;
54	q2 += qd2 * dt_s;
55	q3 += qd3 * dt_s;
56	const float n = sqrtf(q0 * q0 + q1 * q1 + q2 * q2 + q3 * q3);
57	f->q[0] = q0 / n;
58	f->q[1] = q1 / n;
59	f->q[2] = q2 / n;
60	f->q[3] = q3 / n;
61}

63void quat_to_euler_deg(const float q[4], float *roll, float *pitch, float *yaw)
64{
65	const float w = q[0], x = q[1], y = q[2], z = q[3];
66	*roll = atan2f(2.0f * (w * x + y * z), 1.0f - 2.0f * (x * x + y * y)) * RAD2DEG;
67	float sp = 2.0f * (w * y - z * x);
68	sp = sp > 1.0f ? 1.0f : (sp < -1.0f ? -1.0f : sp);
69	*pitch = asinf(sp) * RAD2DEG;
70	*yaw = atan2f(2.0f * (w * z + x * y), 1.0f - 2.0f * (y * y + z * z)) * RAD2DEG;
71}

73float heading_from_yaw_deg(float yaw_deg)
74{
75	float h = fmodf(yaw_deg, 360.0f);
76	return h < 0.0f ? h + 360.0f : h;
77}

79void bias_cal_init(bias_cal_t *c, uint32_t window_samples)
80{
81	memset(c, 0, sizeof *c);
82	c->gyro_thr_dps = 3.0f;
83	c->accel_thr_g = 0.05f;
84	c->window = window_samples ? window_samples : 1u;
85}
86
87void bias_cal_request(bias_cal_t *c)
88{
89	c->requested = true;
90	c->count = 0;
91	c->sum[0] = c->sum[1] = c->sum[2] = 0.0f;
92}

94static bool gyro_quiet(const bias_cal_t *c, const float g[3])
95{
96	return fabsf(g[0]) < c->gyro_thr_dps && fabsf(g[1]) < c->gyro_thr_dps && fabsf(g[2]) < c->gyro_thr_dps;
97}
98
99bool bias_cal_is_at_rest(const bias_cal_t *c, const float gyro_dps[3], const float accel_g[3])
100{
101	const float an = sqrtf(accel_g[0] * accel_g[0] + accel_g[1] * accel_g[1] + accel_g[2] * accel_g[2]);
102	return gyro_quiet(c, gyro_dps) && fabsf(an - 1.0f) < c->accel_thr_g;
103}
104
105bool bias_cal_feed(bias_cal_t *c, const float gyro_dps[3], const float accel_g[3])
106{
107	const bool ok = c->requested ? gyro_quiet(c, gyro_dps) : bias_cal_is_at_rest(c, gyro_dps, accel_g);
108	if (!ok) {
109		c->count = 0;
110		c->sum[0] = c->sum[1] = c->sum[2] = 0.0f;
111		return false;
112	}
113	for (int i = 0; i < 3; i++) {
114		c->sum[i] += gyro_dps[i];
115	}
116	if (++c->count < c->window) {
117		return false;
118	}
119	for (int i = 0; i < 3; i++) {
120		c->bias[i] = c->sum[i] / (float)c->count;
121		c->sum[i] = 0.0f;
122	}
123	c->count = 0;
124	c->valid = true;
125	c->requested = false;
126	c->calibrations++;
127	return true;
128}
129
130void bias_cal_apply(const bias_cal_t *c, const float gyro_in[3], float gyro_out[3])
131{
132	for (int i = 0; i < 3; i++) {
133		gyro_out[i] = gyro_in[i] - (c->valid ? c->bias[i] : 0.0f);
134	}
