1/* Akkuspannung: VBAT_EN an, >= 25 ms warten, SAADC lesen, VBAT_EN aus (Logik in power/vbat.c, Host-getestet). */
2#include <zephyr/kernel.h>
3#include <zephyr/drivers/adc.h>
4#include <zephyr/drivers/gpio.h>
5#include "app.h"
6#include "vbat.h"
7
8#define USER DT_PATH(zephyr_user)
9static const struct adc_dt_spec ch = ADC_DT_SPEC_GET(USER);
10static const struct gpio_dt_spec en = GPIO_DT_SPEC_GET(USER, vbat_en_gpios);
11
12static int z_en(void *c, int on) { (void)c; return gpio_pin_set_dt(&en, on); }
13static void z_sleep(void *c, uint32_t ms) { (void)c; k_msleep((int32_t)ms); }
14static int z_adc(void *c, int32_t *mv)
15{
16	(void)c;
17	int16_t raw;
18	struct adc_sequence seq = {.buffer = &raw, .buffer_size = sizeof raw};
19	int rc = adc_sequence_init_dt(&ch, &seq);
20	if (rc == 0) {
21		rc = adc_read_dt(&ch, &seq);
22	}
23	if (rc == 0) {
24		*mv = raw;
25		rc = adc_raw_to_millivolts_dt(&ch, mv);
26	}
27	return rc;
28}
29
30static const vbat_port_t port = {.ctx = NULL, .set_enable = z_en, .sleep_ms = z_sleep, .read_adc_mv = z_adc};
31
32int battery_init(void)
33{
34	if (!adc_is_ready_dt(&ch) || !gpio_is_ready_dt(&en)) {
35		return -ENODEV;
36	}
37	int rc = gpio_pin_configure_dt(&en, GPIO_OUTPUT_INACTIVE);
38	return rc ? rc : adc_channel_setup_dt(&ch);
39}
40
41static atomic_t last_mv; /* 0 = noch keine gueltige Messung */
42
43uint16_t battery_read_mv(void)
44{
45	uint16_t mv = 0;
46	(void)vbat_measure(&port, DT_PROP(USER, vbat_divider_num), DT_PROP(USER, vbat_divider_den), &mv);
47	atomic_set(&last_mv, mv);
48	return mv;
49}
50
51uint16_t battery_last_mv(void)
52{
53	return (uint16_t)atomic_get(&last_mv);
54}
