1/* 6-Achsen-Sensorfusion (Madgwick, IMU-Variante ohne Magnetometer) und Gyro-Bias-Kalibrierung.
2 * Reines C, ohne Hardwareabhaengigkeit. Einheiten: accel in g, gyro in dps, dt in s. */
3#ifndef MOTION_FUSION_H
4#define MOTION_FUSION_H
5
6#include <stdbool.h>
7#include <stdint.h>
8
9typedef struct {
10	float q[4]; /* w, x, y, z; Einheitsquaternion, Sensor -> Welt */
11	float beta; /* Gradientenschritt, ANNAHME 0.1 */
12} madgwick_t;
13
14void madgwick_init(madgwick_t *f, float beta);
15void madgwick_update(madgwick_t *f, const float gyro_dps[3], const float accel_g[3], float dt_s);
16
17/* Euler in Grad (ZYX: roll um x, pitch um y, yaw um z); yaw in (-180, 180]. */
18void quat_to_euler_deg(const float q[4], float *roll, float *pitch, float *yaw);
19/* Heading in Grad [0, 360) aus yaw. */
20float heading_from_yaw_deg(float yaw_deg);
21
22/* Bias-Kalibrierung: Ruheerkennung (|gyro| und ||a|-1g| unter Schwellen fuer window Samples)
23 * oder erzwungen per Taster lang (dann nur das Gyro-Kriterium, kein Beschleunigungs-Kriterium). */
24typedef struct {
25	float gyro_thr_dps;  /* ANNAHME 3 dps */
26	float accel_thr_g;   /* ANNAHME 0.05 g */
27	uint32_t window;     /* Samples, z. B. 1 s bei ODR */
28	uint32_t count;
29	float sum[3];
30	float bias[3];
31	bool requested;
32	bool valid;
33	uint32_t calibrations;
34} bias_cal_t;
35
36void bias_cal_init(bias_cal_t *c, uint32_t window_samples);
37void bias_cal_request(bias_cal_t *c);
38/* Fuettert ein rohes Sample; gibt true zurueck, wenn gerade ein neuer Bias uebernommen wurde. */
39bool bias_cal_feed(bias_cal_t *c, const float gyro_dps[3], const float accel_g[3]);
40/* Ruhe im letzten Sample (fuer Energiesparen nutzbar). */
41bool bias_cal_is_at_rest(const bias_cal_t *c, const float gyro_dps[3], const float accel_g[3]);
42void bias_cal_apply(const bias_cal_t *c, const float gyro_in[3], float gyro_out[3]);
43
44#endif
