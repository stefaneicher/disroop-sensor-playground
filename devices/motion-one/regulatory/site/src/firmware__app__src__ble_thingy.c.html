26static ssize_t att_status(thingy_status_t s)
27{
28	return s == THINGY_ERR_LEN ? BT_GATT_ERR(BT_ATT_ERR_INVALID_ATTRIBUTE_LEN)
29				   : BT_GATT_ERR(BT_ATT_ERR_VALUE_NOT_ALLOWED);
30}

37static ssize_t rd_motion_cfg(struct bt_conn *c, const struct bt_gatt_attr *a, void *buf, uint16_t len, uint16_t off)
38{
39	uint8_t v[THINGY_MOTION_CONFIG_LEN];
40	k_mutex_lock(&cfg_lock, K_FOREVER);
41	(void)thingy_encode_motion_config(&motion_cfg, v, sizeof v);
42	k_mutex_unlock(&cfg_lock);
43	return bt_gatt_attr_read(c, a, buf, len, off, v, sizeof v);

46static ssize_t wr_motion_cfg(struct bt_conn *c, const struct bt_gatt_attr *a, const void *buf, uint16_t len,
47			     uint16_t off, uint8_t flags)
48{
49	thingy_motion_config_t n;
50	const thingy_status_t s = off ? THINGY_ERR_LEN : thingy_parse_motion_config(buf, len, &n);
51	if (s != THINGY_OK) {
52		return att_status(s);
53	}
54	k_mutex_lock(&cfg_lock, K_FOREVER);
55	motion_cfg = n;
56	k_mutex_unlock(&cfg_lock);
57	LOG_INF("motion_freq_hz=%u", n.motion_freq_hz);
58	return len;
59}

86static ssize_t wr_speaker(struct bt_conn *c, const struct bt_gatt_attr *a, const void *buf, uint16_t len,
87			  uint16_t off, uint8_t flags)
88{
89	thingy_speaker_tone_t t;
90	const thingy_status_t s = off ? THINGY_ERR_LEN : thingy_parse_speaker_tone(buf, len, &t);
91	if (s != THINGY_OK) {
92		return att_status(s);
93	}
94	if (sound_cfg.speaker_mode != 1) {
95		return BT_GATT_ERR(BT_ATT_ERR_VALUE_NOT_ALLOWED); /* nur Frequenzmodus implementiert */
96	}
97	ui_speaker_tone(&t);
98	return len;
99}

279bool ble_motion_wanted(void)
280{
281	for (int i = 0; i < M_COUNT; i++) {
282		if (subscribed(motion_attr[i])) {

289/* Wird mit motion_freq_hz aufgerufen; Dezimierung so, dass Summe der Notifications <= 200/s bleibt. */
290void ble_publish_motion(const float q[4], float roll, float pitch, float yaw, const thingy_raw_motion_t *raw,
291			const thingy_raw_hr_t *hr)
292{
293	static uint32_t tick;
294	bool sub[M_COUNT];
295	uint32_t n = 0;
296	for (int i = 0; i < M_COUNT; i++) {
297		sub[i] = subscribed(motion_attr[i]);
298		n += sub[i];
299	}
300	if (n == 0) {
301		return;
302	}
303	thingy_motion_config_t cfg;
304	ble_get_motion_config(&cfg);
305	const uint32_t want = (uint32_t)cfg.motion_freq_hz * n;
306	const uint32_t every = (want + NOTIFY_BUDGET_PER_S - 1) / NOTIFY_BUDGET_PER_S; /* BERECHNUNG ceil */
307	if ((tick++ % (every ? every : 1)) != 0) {
308		return;
309	}
310	uint8_t b[THINGY_RAW_MOTION_LEN];
311	struct bt_conn *c = conn_cur;
312	if (sub[M_QUAT] && thingy_encode_quaternion(q, b, sizeof b)) {
313		(void)bt_gatt_notify(c, motion_attr[M_QUAT], b, THINGY_QUATERNION_LEN);
314	}
315	if (sub[M_RAW] && thingy_encode_raw_motion(raw, b, sizeof b)) {
316		(void)bt_gatt_notify(c, motion_attr[M_RAW], b, THINGY_RAW_MOTION_LEN);
317	}
318	if (sub[M_EULER] && thingy_encode_euler(roll, pitch, yaw, b, sizeof b)) {
319		(void)bt_gatt_notify(c, motion_attr[M_EULER], b, THINGY_EULER_LEN);
320	}
321	if (sub[M_HEADING] && thingy_encode_heading(yaw, b, sizeof b)) {
322		(void)bt_gatt_notify(c, motion_attr[M_HEADING], b, THINGY_HEADING_LEN);
323	}
324	if (sub[M_HR] && thingy_encode_raw_hr(hr, b, sizeof b)) {
325		(void)bt_gatt_notify(c, motion_attr[M_HR], b, THINGY_RAW_HR_LEN);
326	}
327}
