9void test_layout_vector(void)
10{
11	ui_ext_state_t s = {.flags = UI_EXT_PRESENT | UI_EXT_BTN | UI_EXT_CENTER_OK, .x = -100, .y = 100, .seq = 0xFFFF};
12	uint8_t o[8];
13	const uint8_t want[8] = {0x01, 0x13, 0x9c, 0x64, 0xff, 0xff, 0x00, 0x00};
14	TEST_ASSERT_EQUAL(8, thingy_encode_ui_ext(&s, o, sizeof o));
15	TEST_ASSERT_EQUAL_HEX8_ARRAY(want, o, 8);
16}

18void test_le_seq_and_reserved(void)
19{
20	ui_ext_state_t s = {.flags = UI_EXT_PRESENT, .seq = 0x1234};
21	uint8_t o[8];
22	memset(o, 0xAA, sizeof o);
23	thingy_encode_ui_ext(&s, o, 8);
24	TEST_ASSERT_EQUAL_HEX8(0x34, o[4]);
25	TEST_ASSERT_EQUAL_HEX8(0x12, o[5]);
26	TEST_ASSERT_EQUAL_HEX8(0, o[6]);
27	TEST_ASSERT_EQUAL_HEX8(0, o[7]);
28}

30void test_clamp_and_short_buffer(void)
31{
32	ui_ext_state_t s = {.x = -128, .y = 127};
33	uint8_t o[8];
34	thingy_encode_ui_ext(&s, o, 8);
35	TEST_ASSERT_EQUAL_INT8(-100, (int8_t)o[2]);
36	TEST_ASSERT_EQUAL_INT8(100, (int8_t)o[3]);
37	TEST_ASSERT_EQUAL(0, thingy_encode_ui_ext(&s, o, 7));
38	TEST_ASSERT_EQUAL(0, thingy_encode_ui_ext(NULL, o, 8));
39}

48/* Simulation: Aenderung jede ms, zaehlt Sendungen in 1 s */
49static int run_rate(uint16_t motion_hz)
50{
51	ui_ext_rate_t r;
52	ui_ext_rate_init(&r);
53	int n = 0;
54	for (uint32_t t = 0; t < 1000; t++) {
55		ui_ext_rate_mark(&r, true);
56		n += ui_ext_rate_poll(&r, motion_hz, t, NULL);
57	}
58	return n;
59}

75void test_edge_never_lost_and_within_100ms(void)
76{
77	ui_ext_rate_t r;
78	ui_ext_rate_init(&r);
79	ui_ext_rate_mark(&r, true);
80	TEST_ASSERT_TRUE(ui_ext_rate_poll(&r, 200, 0, NULL));
81	ui_ext_rate_mark(&r, true); /* Flanke 1 ms spaeter, im Sperrintervall */
82	uint32_t retry = 0;
83	TEST_ASSERT_FALSE(ui_ext_rate_poll(&r, 200, 1, &retry));
84	TEST_ASSERT_TRUE(retry > 0 && 1 + retry <= 101);
85	TEST_ASSERT_TRUE(ui_ext_rate_poll(&r, 200, 1 + retry, NULL));
86}
