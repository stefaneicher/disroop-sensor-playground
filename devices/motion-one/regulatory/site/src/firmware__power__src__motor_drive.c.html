3uint16_t motor_supply_mv(uint16_t vbat_mv, bool usb)
4{
5	if (usb) {

7	}
8	if (vbat_mv < MOTOR_SUPPLY_VBAT_MIN_MV || vbat_mv > MOTOR_SUPPLY_VBAT_MAX_MV + 300u) {
9		return MOTOR_SUPPLY_VBAT_MAX_MV;
10	}

14uint16_t motor_duty_permille(uint16_t supply_mv)
15{
16	if (supply_mv == 0) {
17		supply_mv = MOTOR_SUPPLY_USB_MV;
18	}
19	if (supply_mv <= MOTOR_TARGET_MV) {
20		return 1000u;
21	}
22	return (uint16_t)(MOTOR_TARGET_MV * 1000u / supply_mv);
