6static int16_t to_i16_saturated(float value, float scale)
7{
8	const float scaled = roundf(value * scale);
9	if (!(scaled > (float)INT16_MIN)) {
10		return INT16_MIN; /* auch NaN */
11	}
12	if (scaled >= (float)INT16_MAX) {
13		return INT16_MAX;
14	}
15	return (int16_t)scaled;
16}

18static uint8_t *put_i16_le(uint8_t *out, int16_t value)
19{
20	const uint16_t u = (uint16_t)value;
21	out[0] = (uint8_t)(u & 0xFFu);
22	out[1] = (uint8_t)(u >> 8);
23	return out + 2;
24}

26static uint8_t *put_xyz(uint8_t *out, const thingy_xyz_t *v, float scale)
27{
28	out = put_i16_le(out, to_i16_saturated(v->x, scale));
29	out = put_i16_le(out, to_i16_saturated(v->y, scale));
30	return put_i16_le(out, to_i16_saturated(v->z, scale));

44static uint8_t *put_i32_le(uint8_t *out, int32_t value)
45{
46	const uint32_t u = (uint32_t)value;
47	for (int i = 0; i < 4; i++) {
48		out[i] = (uint8_t)(u >> (8 * i));
49	}
50	return out + 4;
51}

53static uint8_t *put_u16_le(uint8_t *out, uint16_t v)
54{
55	out[0] = (uint8_t)(v & 0xFFu);
56	out[1] = (uint8_t)(v >> 8);
57	return out + 2;

60static uint16_t get_u16_le(const uint8_t *in)
61{

226thingy_status_t thingy_parse_led(const uint8_t *in, size_t len, thingy_led_cmd_t *out)
227{
228	if (in == NULL || out == NULL) {
229		return THINGY_ERR_ARG;
230	}
231	if (len < 1) {
232		return THINGY_ERR_LEN;
233	}
234	thingy_led_cmd_t c;
235	memset(&c, 0, sizeof c);
236	c.mode = (thingy_led_mode_t)in[0];
237	switch (in[0]) {
238	case THINGY_LED_OFF:
239		if (len != 1) {
240			return THINGY_ERR_LEN;
241		}
242		break;
243	case THINGY_LED_CONSTANT:
244		if (len != 4) {
245			return THINGY_ERR_LEN;
246		}
247		c.r = in[1];
248		c.g = in[2];
249		c.b = in[3];
250		break;
251	case THINGY_LED_BREATHE:
252		if (len != 5) {
253			return THINGY_ERR_LEN;
254		}
255		c.color_mix = in[1];
256		c.intensity_pct = in[2];
257		c.delay_ms = get_u16_le(&in[3]);
258		if (check_mix_intensity(c.color_mix, c.intensity_pct) != THINGY_OK || c.delay_ms < 50 ||
259		    c.delay_ms > 10000) {
260			return THINGY_ERR_RANGE;
261		}
262		break;
263	case THINGY_LED_ONE_SHOT:
264		if (len != 3) {
265			return THINGY_ERR_LEN;
266		}
267		c.color_mix = in[1];
268		c.intensity_pct = in[2];
269		if (check_mix_intensity(c.color_mix, c.intensity_pct) != THINGY_OK) {
270			return THINGY_ERR_RANGE;
271		}
272		break;
273	default:
274		return THINGY_ERR_RANGE;
275	}
276	*out = c;
277	return THINGY_OK;
278}

280size_t thingy_encode_led(const thingy_led_cmd_t *cmd, uint8_t *out, size_t out_len)
281{
282	if (cmd == NULL || out == NULL) {
283		return 0;
284	}
285	size_t n;
286	switch (cmd->mode) {
287	case THINGY_LED_OFF:
288		n = 1;
289		break;
290	case THINGY_LED_CONSTANT:
291		n = 4;
292		break;
293	case THINGY_LED_BREATHE:
294		n = 5;
295		break;
296	case THINGY_LED_ONE_SHOT:
297		n = 3;
298		break;
299	default:
300		return 0;
301	}
302	if (out_len < n) {
303		return 0;
304	}
305	out[0] = (uint8_t)cmd->mode;
306	if (cmd->mode == THINGY_LED_CONSTANT) {
307		out[1] = cmd->r;
308		out[2] = cmd->g;
309		out[3] = cmd->b;
310	} else if (n >= 3) {
311		out[1] = cmd->color_mix;
312		out[2] = cmd->intensity_pct;
313		if (n == 5) {
314			(void)put_u16_le(&out[3], cmd->delay_ms);
315		}
316	}
317	return n;
318}

320uint8_t thingy_battery_percent_from_mv(uint16_t mv)
321{
322	/* LiPo-Leerlaufkurve, ANNAHME (kein Akku-Datenblatt); 0 % = Abschaltschwelle 3.3 V (BUDGETS) */
323	static const struct {
324		uint16_t mv;
325		uint8_t pct;
326	} k[] = {{3300, 0}, {3600, 10}, {3700, 25}, {3800, 50}, {3900, 65}, {4000, 80}, {4100, 92}, {4200, 100}};
327	const size_t n = sizeof k / sizeof k[0];
328	if (mv <= k[0].mv) {
329		return 0;
330	}
331	for (size_t i = 1; i < n; i++) {
332		if (mv <= k[i].mv) {
333			const uint32_t span = (uint32_t)(k[i].mv - k[i - 1].mv);
334			const uint32_t pos = (uint32_t)(mv - k[i - 1].mv);
335			return (uint8_t)(k[i - 1].pct + (pos * (uint32_t)(k[i].pct - k[i - 1].pct) + span / 2) / span);
336		}
337	}
338	return 100;
339}

341static uint8_t *put_xyz_flags(uint8_t *out, const thingy_xyz_t *v, float scale, uint16_t *flags, unsigned shift)
342{
343	const float in[3] = {v->x, v->y, v->z};
344	for (unsigned i = 0; i < 3; i++) {
345		const int16_t s = to_i16_saturated(in[i], scale);
346		if (s == INT16_MAX || s == INT16_MIN) {
347			*flags |= (uint16_t)(1u << (shift + i));
348		}
349		out = put_i16_le(out, s);
350	}
351	return out;
352}

372thingy_button_evt_t thingy_button_update(thingy_button_t *b, bool pressed, uint32_t now_ms)
373{
374	if (pressed && !b->pressed) {
375		b->pressed = true;
376		b->long_reported = false;
377		b->t_down_ms = now_ms;
378		return THINGY_BTN_EVT_NONE;
379	}
380	if (pressed) {
381		if (!b->long_reported && (uint32_t)(now_ms - b->t_down_ms) >= THINGY_BUTTON_LONG_MS) {
382			b->long_reported = true;
383			return THINGY_BTN_EVT_LONG;
384		}
385		return THINGY_BTN_EVT_NONE;
386	}
387	if (b->pressed) {
388		b->pressed = false;
389		return b->long_reported ? THINGY_BTN_EVT_NONE : THINGY_BTN_EVT_SHORT;
390	}
391	return THINGY_BTN_EVT_NONE;
392}

396size_t thingy_encode_ui_ext(const ui_ext_state_t *s, uint8_t *out, size_t out_len)

423bool ui_ext_rate_poll(ui_ext_rate_t *r, uint16_t motion_freq_hz, uint32_t now_ms, uint32_t *retry_ms)
424{
425	/* 20/s -> 50 ms, Spielmodus (> 50 Hz) 10/s -> 100 ms; Wartezeit <= Intervall <= 100 ms */
426	const uint32_t interval = motion_freq_hz > 50u ? 100u : 50u;
427	if (retry_ms != NULL) {
428		*retry_ms = 0;
429	}
430	if (!r->pending) {
431		return false;
432	}
433	const uint32_t dt = now_ms - r->last_tx_ms;
434	if (r->sent_any && dt < interval) {
435		if (retry_ms != NULL) {
436			*retry_ms = interval - dt;
437		}
438		return false;
439	}
440	r->pending = false;
441	r->sent_any = true;
442	r->last_tx_ms = now_ms;
443	return true;
