1/* Disroop Motion One V1: Thingy:52-kompatible Firmware (nRF52840 + ICM-45686). */
2#include <zephyr/kernel.h>
3#include <zephyr/logging/log.h>
4#include <zephyr/sys/poweroff.h>
5#include <zephyr/bluetooth/services/bas.h>
6#include "app.h"
7
8LOG_MODULE_REGISTER(main, LOG_LEVEL_INF);
9
10#define BATTERY_PERIOD_S 60u
11#define UNDERVOLTAGE_MV 3300u /* BUDGETS, ANNAHME */
12
13static void power_off(const char *why)
14{
15	LOG_WRN("System OFF: %s", why);
16	thingy_led_cmd_t off = {.mode = THINGY_LED_OFF};
17	ui_led_apply(&off);
18	ext_input_suspend();
19	(void)imu_enter_wake_on_motion();
20	(void)ui_arm_button_wakeup();
21	sys_poweroff();
22}
23
24static void update_battery(void)
25{
26	const uint16_t mv = battery_read_mv();
27	if (mv == 0) {
28		return;
29	}
30	(void)bt_bas_set_battery_level(thingy_battery_percent_from_mv(mv)); /* notify nur bei Aenderung */
31	if (mv < UNDERVOLTAGE_MV) {
32		power_off("Unterspannung");
33	}
34}
35
36int main(void)
37{
38	LOG_INF("Motion One V1 start");
39	if (ui_init() || battery_init()) {
40		LOG_ERR("UI/Batterie-Init fehlgeschlagen");
41	}
42	update_battery();
43	if (ble_thingy_init()) {
44		LOG_ERR("BLE-Init fehlgeschlagen");
45		return 0;
46	}
47	if (imu_task_start()) {
48		LOG_ERR("IMU-Init fehlgeschlagen");
49	}
50	if (ext_input_start()) {
51		LOG_ERR("J5-Init fehlgeschlagen");
52	}
53	uint32_t t = 0;
54	for (;;) {
55		k_sleep(K_SECONDS(1));
56		if (++t % BATTERY_PERIOD_S == 0) {
57			update_battery();
58		}
59		if (!ble_is_connected() && imu_at_rest_for_s(REST_SLEEP_S)) {
60			power_off("Ruhe ohne Verbindung");
61		}
62	}
63}
