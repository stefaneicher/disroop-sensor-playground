1#!/usr/bin/env python3
2"""Gleicht electronics/PINOUT_V1.json gegen boards/.../motion_one-pins.dtsi ab.
3Jede DTS-Pinzeile traegt '/* PINOUT:<SIGNAL> */'. Exit 0 nur, wenn alle GPIO-Signale aus dem
4PINOUT genau einmal getaggt sind, Port/Pin stimmen und Polaritaet/Pull passen."""
5import json, re, sys, pathlib
6
7FW = pathlib.Path(__file__).resolve().parents[1]
8pinout = pathlib.Path(sys.argv[1]) if len(sys.argv) > 1 else FW.parent / "electronics/PINOUT_V1.json"
9dtsi = pathlib.Path(sys.argv[2]) if len(sys.argv) > 2 else FW / "boards/disroop/motion_one/motion_one-pins.dtsi"
10# Signale ohne GPIO-Funktion in der App (Hardwarefunktion, siehe dtsi-Kopf)
11NON_GPIO = {"XL1", "XL2", "NRESET", "VBUS", "USB_DM", "USB_DP", "SWDIO", "SWCLK"}
12
13data = json.loads(pinout.read_text())
14want, ain = {}, {}
15for p in data["pins"]:
16    m = re.match(r"P([01])\.(\d+)(?:/AIN(\d))?$", p["gpio"])
17    if p["signal"] in NON_GPIO:
18        continue
19    if not m:
20        sys.exit(f"FAIL unbekanntes GPIO-Format {p['signal']}: {p['gpio']}")
21    want[p["signal"]] = (int(m[1]), int(m[2]))
22    if m[3] is not None:
23        ain[p["signal"]] = int(m[3])
24
25errs, seen = [], {}
26for no, line in enumerate(dtsi.read_text().splitlines(), 1):
27    t = re.search(r"PINOUT:(\w+)", line)
28    if not t:
29        continue
30    sig = t[1]
31    if sig in seen:
32        errs.append(f"{sig}: doppelt getaggt (Zeile {seen[sig]} und {no})")
33    seen[sig] = no
34    if sig not in want:
35        errs.append(f"{sig}: nicht in PINOUT (Zeile {no})"); continue
36    m = re.search(r"NRF_PSEL\(\w+,\s*(\d+),\s*(\d+)\)", line) or re.search(r"&gpio(\d)\s+(\d+)", line)
37    a = re.search(r"NRF_SAADC_AIN(\d)", line)
38    if m:
39        got = (int(m[1]), int(m[2]))
40        if got != want[sig]:
41            errs.append(f"{sig}: DTS P{got[0]}.{got[1]} != PINOUT P{want[sig][0]}.{want[sig][1]} (Zeile {no})")
42    elif a:
43        if ain.get(sig) != int(a[1]):
44            errs.append(f"{sig}: DTS AIN{a[1]} != PINOUT AIN{ain.get(sig)} (Zeile {no})")
45    else:
46        errs.append(f"{sig}: kein Pin in Zeile {no}")
47for sig in sorted(set(want) - set(seen)):
48    errs.append(f"{sig}: im PINOUT, aber nicht in DTS")
49
50text = dtsi.read_text()
51func = {p["signal"]: p["function"].lower() for p in data["pins"]}
52# Polaritaet/Pull aus PINOUT-Funktionstext
53for led in ("LED_R", "LED_G", "LED_B"):
54    if "aktiv low" in func.get(led, "") or any("led aktiv low" in n.lower() for n in data.get("notes", [])):
55        if text.count("PWM_POLARITY_INVERTED") < 3 or "nordic,invert" not in text:
56            errs.append(f"{led}: aktiv low, DTS nicht invertiert"); break
57if "pull-up" in func.get("BTN", "") and not re.search(r"GPIO_PULL_UP \| GPIO_ACTIVE_LOW\)>;\s*/\* PINOUT:BTN", text):
58    errs.append("BTN: interner Pull-up/aktiv low fehlt")
59if "aktiv low" in func.get("IMU_CS", "") and not re.search(r"cs-gpios.*GPIO_ACTIVE_LOW.*PINOUT:IMU_CS", text):
60    errs.append("IMU_CS: nicht aktiv low")
61if "spi-max-frequency = <8000000>" not in text:
62    errs.append("SPI nicht 8 MHz")
63m = re.search(r"VBAT\s*\*\s*(\d+)k/\((\d+)k\+(\d+)k\)", " ".join(data.get("notes", [])))
64if m:
65    num, den = int(m[2]) + int(m[3]), int(m[1])
66    dn = re.search(r"vbat-divider-num = <(\d+)>", text); dd = re.search(r"vbat-divider-den = <(\d+)>", text)
67    if not (dn and dd and int(dn[1]) * den == int(dd[1]) * num):
68        errs.append(f"VBAT-Teiler != {num}/{den}")
69
70print(f"PINOUT {pinout.name} revision {data.get('revision')}: {len(want)} GPIO-Signale, {len(seen)} DTS-Tags")
71for e in errs:
72    print("FAIL", e)
73print("pincheck", "FAIL" if errs else "PASS")
74sys.exit(1 if errs else 0)
