125static void test_led_modes(void)
126{
127	thingy_led_cmd_t c;
128	const uint8_t off[1] = {0};
129	const uint8_t constant[4] = {0x01, 0xFF, 0x10, 0x00};
130	const uint8_t breathe[5] = {0x02, 0x03, 50, 0xE8, 0x03};
131	const uint8_t oneshot[3] = {0x03, 0x04, 100};
132	uint8_t out[5];
133	TEST_ASSERT_EQUAL_INT(THINGY_OK, thingy_parse_led(off, 1, &c));
134	TEST_ASSERT_EQUAL_INT(THINGY_LED_OFF, c.mode);
135	TEST_ASSERT_EQUAL_INT(THINGY_OK, thingy_parse_led(constant, 4, &c));
136	TEST_ASSERT_EQUAL_INT(THINGY_LED_CONSTANT, c.mode);
137	TEST_ASSERT_EQUAL_UINT8(255, c.r);
138	TEST_ASSERT_EQUAL_UINT8(16, c.g);
139	TEST_ASSERT_EQUAL_UINT8(0, c.b);
140	TEST_ASSERT_EQUAL_UINT(4, thingy_encode_led(&c, out, sizeof out));
141	TEST_ASSERT_EQUAL_HEX8_ARRAY(constant, out, 4);
142	TEST_ASSERT_EQUAL_INT(THINGY_OK, thingy_parse_led(breathe, 5, &c));
143	TEST_ASSERT_EQUAL_INT(THINGY_LED_BREATHE, c.mode);
144	TEST_ASSERT_EQUAL_UINT8(3, c.color_mix);
145	TEST_ASSERT_EQUAL_UINT8(50, c.intensity_pct);
146	TEST_ASSERT_EQUAL_UINT16(1000, c.delay_ms);
147	TEST_ASSERT_EQUAL_UINT(5, thingy_encode_led(&c, out, sizeof out));
148	TEST_ASSERT_EQUAL_HEX8_ARRAY(breathe, out, 5);
149	TEST_ASSERT_EQUAL_INT(THINGY_OK, thingy_parse_led(oneshot, 3, &c));
150	TEST_ASSERT_EQUAL_INT(THINGY_LED_ONE_SHOT, c.mode);
151	TEST_ASSERT_EQUAL_UINT8(4, c.color_mix);
152}

191static void test_button_short_and_long(void)
192{

198	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_NONE, thingy_button_update(&b, true, 2799));
199	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_LONG, thingy_button_update(&b, true, 2800));
200	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_NONE, thingy_button_update(&b, true, 3500));
201	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_NONE, thingy_button_update(&b, false, 3600));
202	/* Ueberlauf von uint32 ms */
203	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_NONE, thingy_button_update(&b, true, 0xFFFFFF00u));
204	TEST_ASSERT_EQUAL_INT(THINGY_BTN_EVT_LONG, thingy_button_update(&b, true, 0x00000300u));
