3uint16_t vbat_scale_mv(int32_t adc_mv, uint32_t num, uint32_t den)
4{
5	if (den == 0u || adc_mv <= 0) {
6		return 0;
7	}
8	const uint64_t v = (uint64_t)(uint32_t)adc_mv * num / den;
9	return (uint16_t)(v > UINT16_MAX ? UINT16_MAX : v);
10}

12int vbat_measure(const vbat_port_t *p, uint32_t num, uint32_t den, uint16_t *mv_out)
13{
14	*mv_out = 0;
15	int rc = p->set_enable(p->ctx, 1);
16	if (rc == 0) {
17		p->sleep_ms(p->ctx, VBAT_SETTLE_MS);
18		int32_t mv = 0;
19		rc = p->read_adc_mv(p->ctx, &mv);
20		if (rc == 0) {
21			*mv_out = vbat_scale_mv(mv, num, den);
22		}
23	}
24	(void)p->set_enable(p->ctx, 0); /* Teiler immer abschalten (Ruhestrom) */
25	return rc;
26}
