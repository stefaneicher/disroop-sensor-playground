1/* Host-Tests VBAT-Messsequenz (PCB-Hinweis: VBAT_EN >= 25 ms vor ADC, Teiler 100k/100k -> Faktor 2). */
2#include "unity.h"
3#include "vbat.h"
4
5typedef struct {
6	char log[16];
7	int n;
8	uint32_t slept_ms;
9	int adc_mv;
10	int adc_rc;
11	int en_level;
12} fake_t;
13
14static fake_t fk;
15
16static int f_en(void *c, int on) { fake_t *f = c; f->log[f->n++] = on ? 'E' : 'D'; f->en_level = on; return 0; }
17static void f_sleep(void *c, uint32_t ms) { fake_t *f = c; f->log[f->n++] = 'S'; f->slept_ms += ms; }
18static int f_adc(void *c, int32_t *mv)
19{
20	fake_t *f = c;
21	f->log[f->n++] = 'A';
22	TEST_ASSERT_EQUAL_INT_MESSAGE(1, f->en_level, "ADC ohne VBAT_EN");
23	TEST_ASSERT_TRUE_MESSAGE(f->slept_ms >= VBAT_SETTLE_MS, "Einschwingzeit < 25 ms");
24	*mv = f->adc_mv;
25	return f->adc_rc;
26}
27
28static const vbat_port_t port = {.ctx = &fk, .set_enable = f_en, .sleep_ms = f_sleep, .read_adc_mv = f_adc};
29
30void setUp(void) { fk = (fake_t){.adc_mv = 1900}; }
31void tearDown(void) {}
32
33static void test_settle_constant_at_least_25ms(void) { TEST_ASSERT_TRUE(VBAT_SETTLE_MS >= 25u); }
34
35static void test_sequence_enable_wait_read_disable(void)
36{
37	uint16_t mv = 0;
38	TEST_ASSERT_EQUAL_INT(0, vbat_measure(&port, 2, 1, &mv));
39	TEST_ASSERT_EQUAL_STRING("ESAD", fk.log);
40	TEST_ASSERT_EQUAL_INT(0, fk.en_level);
41}
42
43static void test_divider_factor_two(void)
44{
45	uint16_t mv = 0;
46	fk.adc_mv = 1850; /* 3.70 V Akku */
47	TEST_ASSERT_EQUAL_INT(0, vbat_measure(&port, 2, 1, &mv));
48	TEST_ASSERT_EQUAL_UINT16(3700, mv);
49	TEST_ASSERT_EQUAL_UINT16(4200, vbat_scale_mv(2100, 2, 1));
50	TEST_ASSERT_EQUAL_UINT16(0, vbat_scale_mv(-5, 2, 1));
51	TEST_ASSERT_EQUAL_UINT16(UINT16_MAX, vbat_scale_mv(40000, 2, 1));
52	TEST_ASSERT_EQUAL_UINT16(0, vbat_scale_mv(1000, 2, 0));
53}
54
55static void test_adc_error_still_disables(void)
56{
57	uint16_t mv = 123;
58	fk.adc_rc = -5;
59	TEST_ASSERT_NOT_EQUAL(0, vbat_measure(&port, 2, 1, &mv));
60	TEST_ASSERT_EQUAL_UINT16(0, mv);
61	TEST_ASSERT_EQUAL_INT(0, fk.en_level);
62	TEST_ASSERT_EQUAL_CHAR('D', fk.log[fk.n - 1]);
63}
64
65int main(void)
66{
67	UNITY_BEGIN();
68	RUN_TEST(test_settle_constant_at_least_25ms);
69	RUN_TEST(test_sequence_enable_wait_read_disable);
70	RUN_TEST(test_divider_factor_two);
71	RUN_TEST(test_adc_error_still_disables);
72	return UNITY_END();
73}
