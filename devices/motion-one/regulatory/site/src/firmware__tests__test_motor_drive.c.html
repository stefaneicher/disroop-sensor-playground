1/* Host-Tests F-12: Motor-PWM U_eff <= 3.0 V (LCM0827A3038F, QUELLE Datenblatt) und Arbitrierung Motor/Summer
2 * (ANALYSE_V1 §1a: VBUS-Spitze <= 500 mA nur wenn Motor nicht gleichzeitig mit Summer). */
3#include "unity.h"
4#include "motor_drive.h"
5
6void setUp(void) {}
7void tearDown(void) {}
8
9static uint32_t ueff_mv(uint16_t supply_mv, uint16_t permille) { return (uint32_t)supply_mv * permille / 1000u; }
10
11void test_duty_is_target_over_supply(void)
12{
13	TEST_ASSERT_EQUAL_UINT16(714, motor_duty_permille(4200)); /* 3.0/4.2 */
14	TEST_ASSERT_EQUAL_UINT16(857, motor_duty_permille(3500));
15	TEST_ASSERT_EQUAL_UINT16(616, motor_duty_permille(4870)); /* VSYS an USB, ANALYSE_V1: ~62 % */
16}
17
18void test_duty_full_when_supply_at_or_below_target(void)
19{
20	TEST_ASSERT_EQUAL_UINT16(1000, motor_duty_permille(3000));
21	TEST_ASSERT_EQUAL_UINT16(1000, motor_duty_permille(2900));
22}
23
24void test_ueff_never_above_3v0_over_range(void)
25{
26	for (uint32_t mv = 2500; mv <= 5500; mv += 10) {
27		const uint16_t d = motor_duty_permille((uint16_t)mv);
28		TEST_ASSERT_TRUE(d <= 1000);
29		TEST_ASSERT_TRUE_MESSAGE(ueff_mv((uint16_t)mv, d) <= MOTOR_TARGET_MV, "U_eff > 3.0 V");
30	}
31}
32
33void test_duty_zero_supply_uses_conservative_default(void)
34{
35	TEST_ASSERT_EQUAL_UINT16(motor_duty_permille(MOTOR_SUPPLY_USB_MV), motor_duty_permille(0));
36}
37
38void test_supply_battery_measured(void)
39{
40	TEST_ASSERT_EQUAL_UINT16(3700, motor_supply_mv(3700, false));
41}
42
43void test_supply_no_or_implausible_measurement_uses_vbat_max(void)
44{
45	TEST_ASSERT_EQUAL_UINT16(MOTOR_SUPPLY_VBAT_MAX_MV, motor_supply_mv(0, false));
46	TEST_ASSERT_EQUAL_UINT16(MOTOR_SUPPLY_VBAT_MAX_MV, motor_supply_mv(1200, false));
47	TEST_ASSERT_EQUAL_UINT16(MOTOR_SUPPLY_VBAT_MAX_MV, motor_supply_mv(6000, false));
48}
49
50void test_supply_usb_uses_vsys_worst_case(void)
51{
52	/* An USB haengt VSW an VSYS (bis 4.87 V SIMULATION); VBAT-Messung waere zu tief */
53	TEST_ASSERT_EQUAL_UINT16(MOTOR_SUPPLY_USB_MV, motor_supply_mv(3700, true));
54	TEST_ASSERT_TRUE(MOTOR_SUPPLY_USB_MV >= 4870);
55}
56
57void test_arbitration_motor_alone(void)
58{
59	load_out_t o = load_arbitrate(false, true);
60	TEST_ASSERT_TRUE(o.motor_on);
61	TEST_ASSERT_FALSE(o.buzzer_on);
62}
63
64void test_arbitration_buzzer_alone(void)
65{
66	load_out_t o = load_arbitrate(true, false);
67	TEST_ASSERT_TRUE(o.buzzer_on);
68	TEST_ASSERT_FALSE(o.motor_on);
69}
70
71void test_arbitration_buzzer_pauses_motor(void)
72{
73	load_out_t o = load_arbitrate(true, true);
74	TEST_ASSERT_TRUE(o.buzzer_on);
75	TEST_ASSERT_FALSE_MESSAGE(o.motor_on, "Motor und Summer gleichzeitig");
76}
77
78void test_arbitration_none(void)
79{
80	load_out_t o = load_arbitrate(false, false);
81	TEST_ASSERT_FALSE(o.buzzer_on || o.motor_on);
82}
83
84void test_vbus_peak_within_usb_limit(void)
85{
86	/* ANALYSE_V1 §1a: Lader 281 + CHRG-LED 3 + 3V3 138.6 (inkl. Summer 87) mA; Motor-Anlauf 120 mA (QUELLE DB) */
87	TEST_ASSERT_TRUE(motor_vbus_peak_ma(true, false) <= 500u);
88	TEST_ASSERT_TRUE(motor_vbus_peak_ma(false, true) <= 500u);
89	TEST_ASSERT_TRUE(motor_vbus_peak_ma(true, true) > 500u); /* darum Arbitrierung */
90	load_out_t o = load_arbitrate(true, true);
91	TEST_ASSERT_TRUE(motor_vbus_peak_ma(o.buzzer_on, o.motor_on) <= 500u);
92}
