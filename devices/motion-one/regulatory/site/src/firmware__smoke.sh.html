1#!/usr/bin/env bash
2# Firmware-Smoke (PROMPT_1 §5.3/§9). Laeuft im Image disroop-motion-one-fw:
3#   ./project hw fw "bash devices/motion-one/firmware/smoke.sh [outdir]"
4# 1. alle Host-Unit-Tests (gcc + Unity)  2. Zephyr-App fuer Board motion_one  3. Images + SHA-256
5set -uo pipefail
6cd "$(dirname "${BASH_SOURCE[0]}")"
7FW="$(pwd)"
8OUT="$(cd ../../.. && pwd)/${1:-devices/motion-one/logs/fw-v1}"
9mkdir -p "$OUT"
10rc=0
11
12{
13  echo "date: $(date -Iseconds)"
14  echo "gcc: $(gcc --version | head -1)"
15  echo "cmake: $(cmake --version | head -1)"
16  echo "west: $(west --version)"
17  echo "zephyr: $(tr '\n' ' ' < "$ZEPHYR_BASE/VERSION")"
18  echo "sdk: $(cat "$ZEPHYR_SDK_INSTALL_DIR/sdk_version")"
19  echo "unity: $(grep -E 'define UNITY_VERSION_(MAJOR|MINOR|BUILD)' "$UNITY_DIR/src/unity.h" | awk '{print $3}' | paste -sd.)"
20} > "$OUT/versions.txt" 2>&1
21
22echo "== Host-Unit-Tests"
23make -s clean && make test > "$OUT/unit-tests.log" 2>&1 || rc=1
24grep -E "^(==|[0-9]+ Tests)" "$OUT/unit-tests.log"
25grep -E "^[0-9]+ Tests" "$OUT/unit-tests.log" | grep -vq " 0 Failures" && rc=1
26echo "Summe: $(grep -E '^[0-9]+ Tests' "$OUT/unit-tests.log" | awk '{t+=$1; f+=$3} END {print t" Tests, "f" Failures"}')"
27
28echo "== Zephyr-App motion_one"
29bash "$FW/build-app.sh" > "$OUT/app-build.log" 2>&1 || rc=1
30grep -E "warning|error:|FLASH:|RAM:" "$OUT/app-build.log"
31grep -q "warning:" "$OUT/app-build.log" && { echo "Warnungen im App-Build"; rc=1; }
32Z="$FW/build/app/zephyr"
33if [[ -f "$Z/zephyr.uf2" && -f "$Z/zephyr.hex" ]]; then
34  cp "$Z/zephyr.uf2" "$Z/zephyr.hex" "$Z/zephyr.elf" "$Z/zephyr.bin" "$OUT/"
35  (cd "$OUT" && sha256sum zephyr.uf2 zephyr.hex zephyr.bin zephyr.elf > zephyr.sha256 && ls -l zephyr.* >> zephyr.sha256)
36  file "$OUT/zephyr.elf" >> "$OUT/zephyr.sha256"
37  echo "hex-start: $(head -2 "$OUT/zephyr.hex" | tr '\n' ' ')" >> "$OUT/zephyr.sha256"
38  cat "$OUT/zephyr.sha256"
39else
40  echo "zephyr.uf2/hex fehlt"; rc=1
41fi
42echo "smoke rc=$rc"
43exit $rc
