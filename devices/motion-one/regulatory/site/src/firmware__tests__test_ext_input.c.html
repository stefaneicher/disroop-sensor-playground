5static stick_axis_cal_t c;
6void setUp(void) { stick_axis_cal_init(&c); }
7void tearDown(void) {}

9void test_scale_default_center_and_ends(void)
10{
11	TEST_ASSERT_EQUAL_INT8(0, stick_axis_scale(&c, 1650, 0));
12	TEST_ASSERT_EQUAL_INT8(100, stick_axis_scale(&c, 3300, 0));
13	TEST_ASSERT_EQUAL_INT8(-100, stick_axis_scale(&c, 0, 0));
14	TEST_ASSERT_EQUAL_INT8(50, stick_axis_scale(&c, 2475, 0));
15	TEST_ASSERT_EQUAL_INT8(100, stick_axis_scale(&c, 3600, 0)); /* geklemmt */
16}
17
18void test_deadzone(void)
19{
20	TEST_ASSERT_EQUAL_INT8(0, stick_axis_scale(&c, 1650 + 120, 8)); /* 7 % */
21	TEST_ASSERT_EQUAL_INT8(0, stick_axis_scale(&c, 1650 - 120, 8));
22	TEST_ASSERT_TRUE(stick_axis_scale(&c, 1650 + 200, 8) > 0);   /* 12 % */
23	TEST_ASSERT_EQUAL_INT8(100, stick_axis_scale(&c, 3300, 8));  /* Endwert bleibt 100 */
24	TEST_ASSERT_EQUAL_INT8(-100, stick_axis_scale(&c, 0, 8));
25}

27void test_center_calibration_shifts_zero(void)
28{
29	const uint16_t s[4] = {1790, 1810, 1800, 1800};
30	TEST_ASSERT_TRUE(stick_axis_cal_center(&c, s, 4));
31	TEST_ASSERT_EQUAL_UINT16(1800, c.center_mv);
32	TEST_ASSERT_EQUAL_INT8(0, stick_axis_scale(&c, 1800, 0));
33	TEST_ASSERT_EQUAL_INT8(100, stick_axis_scale(&c, 3300, 0));
34	TEST_ASSERT_EQUAL_INT8(-100, stick_axis_scale(&c, 0, 0));
35	TEST_ASSERT_EQUAL_INT8(-50, stick_axis_scale(&c, 900, 0)); /* asymmetrisch skaliert */
36}

38void test_center_calibration_rejects_implausible(void)
39{
40	const uint16_t s[2] = {3200, 3300}; /* ausgelenkt oder offen */

42	TEST_ASSERT_EQUAL_UINT16(1650, c.center_mv);
43	TEST_ASSERT_FALSE(stick_axis_cal_center(&c, s, 0));

59void test_poll_on_demand(void)
60{

69void test_touch_debounced_events(void)
70{
71	touch_t t;
72	touch_init(&t);
73	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 100));
74	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 110)); /* < 30 ms */
75	TEST_ASSERT_EQUAL(TOUCH_EVT_DOWN, touch_update(&t, true, 130));
76	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 200));
77	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, false, 500));
78	TEST_ASSERT_EQUAL(TOUCH_EVT_UP, touch_update(&t, false, 530));
79}

90void test_touch_floating_input_no_spam(void)
91{
92	touch_t t;
93	touch_init(&t);
94	int events = 0;
95	for (uint32_t ms = 0; ms < 30000; ms += 20) { /* 25 Hz Flattern ueber 30 s, je 20 ms stabil */
96		events += touch_update(&t, (ms / 40) % 2, ms) != TOUCH_EVT_NONE;
97		events += touch_update(&t, (ms / 40) % 2, ms + 19) != TOUCH_EVT_NONE;
98	}
99	TEST_ASSERT_TRUE_MESSAGE(t.locked, "Flattern nicht erkannt");
100	TEST_ASSERT_TRUE_MESSAGE(events <= (int)TOUCH_SPAM_EDGES, "Ereignisspam bei offenem Eingang"); /* begrenzt statt 750 */
101}

103void test_touch_recovers_after_lockout(void)
104{
105	touch_t t;
106	touch_init(&t);
107	for (uint32_t ms = 0; ms < 1000; ms += 10) {
108		touch_update(&t, (ms / 10) % 2, ms);
109	}
110	TEST_ASSERT_TRUE(t.locked);
111	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 5000));  /* noch gesperrt */
112	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 12000)); /* 10 s ohne Flanke: stilles Entsperren */
113	TEST_ASSERT_FALSE(t.locked);
114	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, false, 12100));
115	TEST_ASSERT_EQUAL(TOUCH_EVT_UP, touch_update(&t, false, 12130));
116	TEST_ASSERT_EQUAL(TOUCH_EVT_NONE, touch_update(&t, true, 12200));
117	TEST_ASSERT_EQUAL(TOUCH_EVT_DOWN, touch_update(&t, true, 12230));
118}
