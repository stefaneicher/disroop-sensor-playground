25static int bus_write(void *ctx, uint8_t reg, const uint8_t *d, size_t len)
26{
27	ARG_UNUSED(ctx);
28	const struct spi_buf b[2] = {{.buf = &reg, .len = 1}, {.buf = (void *)d, .len = len}};
29	const struct spi_buf_set tx = {b, 2};
30	return spi_write_dt(&spi, &tx);
31}
32
33static int bus_read(void *ctx, uint8_t reg, uint8_t *d, size_t len)
34{
35	ARG_UNUSED(ctx);
36	uint8_t cmd = reg | ICM_SPI_READ_BIT;
37	const struct spi_buf tb = {.buf = &cmd, .len = 1};
38	const struct spi_buf rb[2] = {{.buf = NULL, .len = 1}, {.buf = d, .len = len}};
39	const struct spi_buf_set tx = {&tb, 1}, rx = {rb, 2};
40	return spi_transceive_dt(&spi, &tx, &rx);
41}

43static void bus_delay(void *ctx, uint32_t us)
44{
45	ARG_UNUSED(ctx);
46	k_usleep(us);
47}

67int imu_enter_wake_on_motion(void)
68{
69	if (icm45686_enter_wom(&bus, WOM_THRESHOLD_MG) != ICM_OK) {
70		return -EIO;

72	/* nRF52 System OFF: GPIO-SENSE weckt bei aktivem Pegel */
73	return gpio_pin_interrupt_configure_dt(&int1, GPIO_INT_LEVEL_ACTIVE);

76static void process(const icm_sample_t *s, uint32_t decim)
77{
78	float g[3];
79	const bool fresh = bias_cal_feed(&cal, s->gyro_dps, s->accel_g);
80	if (fresh) {
81		LOG_INF("Gyro-Bias neu (%d)", (int)cal.calibrations);
82	}
83	rest_samples = bias_cal_is_at_rest(&cal, s->gyro_dps, s->accel_g) ? rest_samples + 1 : 0;
84	bias_cal_apply(&cal, s->gyro_dps, g);
85	madgwick_update(&fus, g, s->accel_g, 1.0f / (float)IMU_ODR_HZ);
86	sample_counter++;
87	if (sample_counter % decim != 0 || !ble_motion_wanted()) {
88		return;
89	}
90	float r, p, y;
91	quat_to_euler_deg(fus.q, &r, &p, &y);
92	const thingy_raw_motion_t raw = {{s->accel_g[0], s->accel_g[1], s->accel_g[2]},
93					 {g[0], g[1], g[2]}, {0, 0, 0}};
94	thingy_raw_hr_t hr = {raw.accel, raw.gyro, sample_counter, cal.valid ? THINGY_HR_FLAG_FUSION_OK : 0};
95	for (int i = 0; i < 3; i++) {
96		hr.flags |= (uint16_t)((s->accel_sat[i] ? THINGY_HR_FLAG_ACC_SAT_X << i : 0) |
97				       (s->gyro_sat[i] ? THINGY_HR_FLAG_GYR_SAT_X << i : 0));
98	}
99	ble_publish_motion(fus.q, r, p, y, &raw, &hr);
100}

102static void imu_thread(void *a, void *b, void *c)
103{
104	static uint8_t buf[MAX_FRAMES * ICM_FIFO_FRAME_LEN];
105	for (;;) {
106		(void)k_sem_take(&data_sem, K_MSEC(100)); /* Timeout als Rueckfallebene, falls INT verpasst */
107		thingy_motion_config_t cfg;
108		ble_get_motion_config(&cfg);
109		const uint32_t decim = IMU_ODR_HZ / cfg.motion_freq_hz ? IMU_ODR_HZ / cfg.motion_freq_hz : 1;
110		uint16_t n = 0;
111		if (icm45686_fifo_count(&bus, &n) != ICM_OK || n == 0) {
112			continue;
113		}
114		n = MIN(n, MAX_FRAMES);
115		if (bus.read(NULL, ICM_REG_FIFO_DATA, buf, n * ICM_FIFO_FRAME_LEN)) {
116			continue;
117		}
118		for (uint16_t i = 0; i < n; i++) {
119			icm_sample_t s;
120			if (icm45686_parse_frame(&buf[i * ICM_FIFO_FRAME_LEN], ICM_FIFO_FRAME_LEN, &s) == ICM_FRAME_OK) {
121				process(&s, decim);
122			}
123		}
124	}
125}

128static struct k_thread imu_thr;

130int imu_task_start(void)
131{
132	bias_cal_init(&cal, IMU_ODR_HZ); /* 1 s Ruhe */
133	madgwick_init(&fus, 0.1f);
134	if (!spi_is_ready_dt(&spi) || !gpio_is_ready_dt(&int1)) {
135		return -ENODEV;
136	}
137	const icm_status_t st = icm45686_init(&bus, ICM_ODR_400HZ, WATERMARK_FRAMES);
138	if (st != ICM_OK) {
139		LOG_ERR("ICM-45686 init %d", st);
140		return -EIO;
141	}
142	gpio_pin_configure_dt(&int1, GPIO_INPUT);
143	gpio_init_callback(&int1_cb, int1_isr, BIT(int1.pin));
144	gpio_add_callback(int1.port, &int1_cb);
145	gpio_pin_interrupt_configure_dt(&int1, GPIO_INT_EDGE_TO_ACTIVE);
146	k_thread_create(&imu_thr, imu_stack, K_THREAD_STACK_SIZEOF(imu_stack), imu_thread, NULL, NULL, NULL,
147			K_PRIO_PREEMPT(2), 0, K_NO_WAIT);
148	k_thread_name_set(&imu_thr, "imu");
149	return 0;
